# match/case structural pattern matching — Python 3.10+
def classify_response(response):
    match response:
        case {"status": 200, "data": data}:
            return f"Success: {data}"
        case {"status": 404}:
            return "Not found"
        case {"status": code} if code >= 500:  # guard clause
            return f"Server error: {code}"
        case _:
            return "Unknown"

# for/else — the else block runs only if no break occurred
def find_prime(candidates):
    for n in candidates:
        if is_prime(n):
            print(f"Found: {n}")
            break
    else:
        print("No primes in list")  # runs if no break

functions.py
# Positional-only (/) and keyword-only (*) — Python 3.8+
def format_name(first, last, /, *, separator=" ", upper=False):
    """first and last are positional-only (before /).
    separator, upper are keyword-only (after *).
    """
    result = f"{first}{separator}{last}"
    return result.upper() if upper else result

format_name("Ravi", "Kumar")                       # OK
format_name("Ravi", "Kumar", separator="_")        # OK
# format_name(first="Ravi", last="Kumar")          # TypeError!

# Closure — function that remembers its creation context
def make_counter(start=0, step=1):
    count = start
    def counter():
        nonlocal count   # modify the enclosing scope
        val = count
        count += step
        return val
    return counter

c = make_counter(start=10, step=5)
print(c(), c(), c())  # → 10 15 20

bank_account.py
class BankAccount:
    interest_rate = 0.05          # class attribute — shared

    def __init__(self, owner: str, balance: float = 0) -> None:
        self.owner = owner
        self.__balance = balance   # name-mangled → _BankAccount__balance

    @property
    def balance(self) -> float:
        return self.__balance

    @balance.setter
    def balance(self, value: float) -> None:
        if value < 0:
            raise ValueError(f"Balance cannot be negative: {value}")
        self.__balance = value

    @classmethod
    def from_dict(cls, data: dict) -> "BankAccount":
        return cls(data["owner"], data.get("balance", 0))

    @staticmethod
    def is_valid_owner(name: str) -> bool:
        return bool(name) and name.isalpha()
